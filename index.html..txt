  if(saiSeconds<=0)saiSeconds=25*60;

function calcStreak(){
 var s=getStreak();
 var today=td();
 var cur=s.current||0;
 var last=s.lastDate||'';
 var best=s.best||0;

 // حساب الـ streak بناءً على آخر يوم تم فيه بوح أو ورد أو مهمة
 var yest=new Date();yest.setDate(yest.getDate()-1);
 var yestStr=fds(yest);

 // نعتبر اليوم "نشط" لو فيه أي مهمة أُنجزت أو ورد أو بوح
 var todayActive=false;
 if(_db.daily&&_db.daily[today]){
  var d=_db.daily[today];
  if(d.tasks&&Object.keys(d.tasks).some(function(k){return d.tasks[k];}))todayActive=true;
  if(d.thoughts&&d.thoughts.trim())todayActive=true;
 }
 if(_db.drHabits&&_db.drHabits[today])todayActive=true;
 if(_db.drRefs&&_db.drRefs[today]&&Object.values(_db.drRefs[today]).some(function(v){return v&&v.trim();}))todayActive=true;

 if(todayActive){
  if(last===today){
   // نفس اليوم — لا تغيير
  } else if(last===yestStr){
   cur=cur+1;
  } else if(!last){
   cur=1;
  } else {
   cur=1;
  }
  s.lastDate=today;
  s.current=cur;
  s.best=Math.max(best,cur);
  dbSv();
 } else if(last&&last!==today&&last!==yestStr){
  // انكسر الـ streak
  s.current=0;
  s.lastDate='';
  dbSv();
  cur=0;
 }
 return {current:cur,best:s.best||0};
}

var taskDragId=null;
function initTaskDrag(){
 var items=document.querySelectorAll('#ul-tasks .ti[draggable]');
 items.forEach(function(el){
  el.addEventListener('dragstart',function(e){
   taskDragId=this.dataset.id;
   setTimeout(function(){el.style.opacity='.35';},0);
   e.dataTransfer.effectAllowed='move';
  });
  el.addEventListener('dragend',function(){
   this.style.opacity='';
   document.querySelectorAll('.ti.drag-over-task').forEach(function(x){x.classList.remove('drag-over-task');});
   taskDragId=null;
  });
  el.addEventListener('dragover',function(e){
   e.preventDefault();
   if(this.dataset.id===taskDragId)return;
   document.querySelectorAll('.ti.drag-over-task').forEach(function(x){x.classList.remove('drag-over-task');});
   this.classList.add('drag-over-task');
  });
  el.addEventListener('dragleave',function(){this.classList.remove('drag-over-task');});
  el.addEventListener('drop',function(e){
   e.preventDefault();
   this.classList.remove('drag-over-task');
   if(!taskDragId||this.dataset.id===taskDragId)return;
   var tasks=getTasks();
   var fromIdx=tasks.findIndex(function(t){return String(t.id)===taskDragId;});
   var toIdx=tasks.findIndex(function(t){return String(t.id)===this.dataset.id;}.bind(this));
   if(fromIdx<0||toIdx<0)return;
   var moved=tasks.splice(fromIdx,1)[0];
   tasks.splice(toIdx,0,moved);
   setTasks(tasks);renderTasks();
  });
 });
}

function initKeyboardShortcuts(){
 document.addEventListener('keydown',function(e){
  /* لو في input/textarea — لا تشتغل */
  var tag=document.activeElement&&document.activeElement.tagName;
  if(tag==='INPUT'||tag==='TEXTAREA'||tag==='SELECT')return;
  /* لو مودال مفتوح — سيب Escape فقط */
  var modal=document.querySelector('.modal-bg.open');

  switch(e.key){
   case 'n': case 'N':
    /* N — أضف مهمة جديدة */
    if(!modal){
     e.preventDefault();
     var activeTab=document.querySelector('.tb.on');
     if(activeTab&&activeTab.dataset.t==='today'){
      document.getElementById('add-task-form').style.display='flex';
      document.getElementById('inp-task').focus();
     } else if(activeTab&&activeTab.dataset.t==='plan'){
      document.getElementById('btn-add-plan-task').click();
     }
    }
    break;
   case '1':
    /* 1 — تبويب اليوم */
    if(!modal){e.preventDefault();document.querySelector('.tb[data-t="plan"]').click();}
    break;
   case '2':
    /* 2 — تبويب اليوم */
    if(!modal){e.preventDefault();document.querySelector('.tb[data-t="today"]').click();}
    break;

/* ═══ Streak: تعريف أوضح للنشاط ═══ */
/* تم تحديث calcStreak — النشاط = إنجاز مهمة واحدة على الأقل أو ورد الليلة */
function calcStreak(){
 dbLoad();
 if(!_db.streak)_db.streak={};
 var s=_db.streak;
 var today=td();
 var cur=s.current||0;
 var last=s.lastDate||'';
 var best=s.best||0;
 var yest=new Date();yest.setDate(yest.getDate()-1);
 var yestStr=fds(yest);

 /* نشاط حقيقي = مهمة منجزة OR ورد الليلة */
 var todayActive=false;
 var tasks=gp('daily.'+today+'.tasks')||{};
 if(Object.values(tasks).some(function(v){return v===true||v===1;}))todayActive=true;
 if(_db.drHabits&&_db.drHabits[today])todayActive=true;

 if(todayActive){
  if(last===today){/* نفس اليوم */}
  else if(last===yestStr){cur++;}
  else{cur=1;}
  s.lastDate=today;s.current=cur;s.best=Math.max(best,cur);
  dbSv();
 } else if(last&&last!==today&&last!==yestStr){
  s.current=0;s.lastDate='';dbSv();cur=0;
 }
 return {current:cur,best:s.best||0};
}
